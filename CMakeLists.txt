cmake_minimum_required(VERSION 3.22)

project(box2d-samples LANGUAGES C CXX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${PROJECT_SOURCE_DIR}/bin/Debug")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO "${PROJECT_SOURCE_DIR}/bin/RelWithDebInfo")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${PROJECT_SOURCE_DIR}/bin/Release")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${PROJECT_SOURCE_DIR}/libs/Debug")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELWITHDEBINFO "${PROJECT_SOURCE_DIR}/libs/RelWithDebInfo")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${PROJECT_SOURCE_DIR}/libs/Release")

include(cmake/Dep_Box2D.cmake)
include(cmake/Dep_GLFW.cmake)
include(cmake/Dep_GLAD.cmake)
include(cmake/Dep_IMGUI.cmake)
include(cmake/Dep_IMPLOT.cmake)
include(cmake/Dep_NFD.cmake)
include(cmake/Dep_JSMN.cmake)

set(SHADER_FILES
    data/background.vs
    data/background.fs
    data/circle.vs
    data/circle.fs
    data/line.vs
    data/line.fs
    data/point.vs
    data/point.fs
    data/solid_capsule.vs
    data/solid_capsule.fs
    data/solid_circle.vs
    data/solid_circle.fs
    data/solid_polygon.vs
    data/solid_polygon.fs
)

set(SHADER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/shaders_embedded.h")
file(WRITE "${SHADER_HEADER}" "// Generated by CMake from src/data/*.vs and *.fs. Do not edit.\n#pragma once\n\n")
foreach(SHADER_FILE IN LISTS SHADER_FILES)
    set(SHADER_PATH "${PROJECT_SOURCE_DIR}/src/${SHADER_FILE}")
    file(READ "${SHADER_PATH}" SHADER_HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," SHADER_BYTES "${SHADER_HEX}")
    get_filename_component(SHADER_NAME "${SHADER_FILE}" NAME)
    string(REPLACE "." "_" SHADER_SYMBOL "${SHADER_NAME}")
    file(APPEND "${SHADER_HEADER}" "static const char k_${SHADER_SYMBOL}[] = { ${SHADER_BYTES} 0x00 };\n\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHADER_PATH}")
endforeach()

set(BOX2D_SHARED_FILES
    src/shared/benchmarks.c
    src/shared/benchmarks.h
    src/shared/determinism.c
    src/shared/determinism.h
    src/shared/human.c
    src/shared/human.h
    src/shared/utils.c
    src/shared/utils.h
)

add_library(shared STATIC ${BOX2D_SHARED_FILES})
set_target_properties(shared PROPERTIES
    C_STANDARD 17
    C_STANDARD_REQUIRED YES
)
target_include_directories(shared PUBLIC "${PROJECT_SOURCE_DIR}/src/shared")
target_link_libraries(shared PUBLIC box2d)

set(BOX2D_SAMPLE_FILES
    src/car.cpp
    src/car.h
    src/container.c
    src/container.h
    src/donut.cpp
    src/donut.h
    src/doohickey.cpp
    src/doohickey.h
    src/draw.c
    src/draw.h
    src/draw_text.cpp
    src/main.cpp
    src/sample.cpp
    src/sample.h
    src/sample_benchmark.cpp
    src/sample_bodies.cpp
    src/sample_character.cpp
    src/sample_collision.cpp
    src/sample_continuous.cpp
    src/sample_determinism.cpp
    src/sample_events.cpp
    src/sample_geometry.cpp
    src/sample_issues.cpp
    src/sample_joints.cpp
    src/sample_replay.cpp
    src/sample_robustness.cpp
    src/sample_shapes.cpp
    src/sample_stacking.cpp
    src/sample_world.cpp
    src/shader.c
    src/shader.h
    src/stb_image_write.h
)

add_executable(samples ${BOX2D_SAMPLE_FILES})
set_target_properties(samples PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS NO
)
target_include_directories(samples PRIVATE
    "${PROJECT_SOURCE_DIR}/src"
    "${PROJECT_SOURCE_DIR}/src/shared"
    "${CMAKE_CURRENT_BINARY_DIR}"
)
target_link_libraries(samples PRIVATE
    box2d
    shared
    glfw
    glad
    imgui
    implot
    nfd
    jsmn
)

source_group(TREE "${PROJECT_SOURCE_DIR}/src" PREFIX "src" FILES ${BOX2D_SAMPLE_FILES})
source_group(TREE "${PROJECT_SOURCE_DIR}/src/shared" PREFIX "src/shared" FILES ${BOX2D_SHARED_FILES})

if(MSVC)
    set_property(DIRECTORY "${PROJECT_SOURCE_DIR}" PROPERTY VS_STARTUP_PROJECT samples)
    set_property(TARGET samples PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
endif()
